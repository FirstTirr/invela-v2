import React, { useState } from 'react';
import PageAnimateWrapper from '@/components/page-animate-wrapper';
import { Plus, Edit2, Trash2, X } from 'lucide-react';
import { motion, AnimatePresence } from 'framer-motion';

interface LaborItem {
  id: string;
  nama: string;
  jurusan: string;
  labor: string;
  jumlah: number;
  kategori: string;
  status: 'Baik' | 'Perbaikan' | 'Rusak';
}

export default function KabengItemsPage() {
  const [isOpenAddModal, setIsOpenAddModal] = useState(false);
  const [isOpenEditModal, setIsOpenEditModal] = useState(false);
  const [selectedItem, setSelectedItem] = useState<LaborItem | null>(null);

  const [items, setItems] = useState<LaborItem[]>([
    { id: 'BRG-01', nama: 'PC Client Asus ExpertCenter', jurusan: 'PPLG', labor: 'Labor Komputer Pemrograman', jumlah: 36, kategori: 'Perangkat Keras Utama', status: 'Baik' },
    { id: 'BRG-02', nama: 'Drawing Tablet Wacom Intuos', jurusan: 'DKV', labor: 'Labor Multimedia & DKV', jumlah: 15, kategori: 'Alat Pendukung Pengembang', status: 'Perbaikan' }
  ]);

  const handleDelete = (id: string) => {
    setItems(items.filter(item => item.id !== id));
  };

  const handleEditClick = (item: LaborItem) => {
    setSelectedItem(item);
    setIsOpenEditModal(true);
  };

  return (
    <PageAnimateWrapper>
      <div className="space-y-6 font-sans antialiased tracking-tight">
        {/* Header Responsif */}
        <div className="flex flex-col sm:flex-row sm:justify-between sm:items-center gap-4 border-b border-surface-container pb-4">
          <div>
            <h1 className="text-3xl font-bold tracking-tight text-on-surface">Manajemen & Registrasi Barang</h1>
            <p className="text-base text-on-surface-variant mt-1 font-medium">Input data barang baru dan kelola aset ruangan labor.</p>
          </div>
          <button 
            onClick={() => setIsOpenAddModal(true)}
            className="px-5 py-2.5 text-sm font-bold text-white bg-primary hover:bg-primary-container rounded-lg flex items-center justify-center gap-2 shadow-sm transition-all cursor-pointer w-full sm:w-auto"
          >
            <Plus className="w-5 h-5" /> Input Barang Baru
          </button>
        </div>

        {/* Inventory Table Container - Wrapper Scrollable Mulus */}
        <div className="bg-white border border-surface-container-high rounded-xl overflow-hidden shadow-sm w-full">
          <div className="overflow-x-auto w-full">
            <table className="w-full text-left border-collapse text-base min-w-[900px]">
              <thead className="bg-surface-low border-b border-surface-container-high">
                <tr>
                  <th className="p-5 text-sm font-bold text-on-surface tracking-wide uppercase">Nama Barang</th>
                  <th className="p-5 text-sm font-bold text-on-surface tracking-wide uppercase">Klaster Labor</th>
                  <th className="p-5 text-sm font-bold text-on-surface tracking-wide uppercase text-center">Jumlah</th>
                  <th className="p-5 text-sm font-bold text-on-surface tracking-wide uppercase">Kategori</th>
                  <th className="p-5 text-sm font-bold text-on-surface tracking-wide uppercase text-center">Status</th>
                  <th className="p-5 text-sm font-bold text-on-surface tracking-wide uppercase text-right">Aksi</th>
                </tr>
              </thead>
              <tbody className="text-on-surface font-semibold relative">
                <AnimatePresence initial={false}>
                  {items.map((item) => (
                    <motion.tr 
                      key={item.id}
                      layout
                      initial={{ opacity: 0, height: 0 }}
                      animate={{ opacity: 1, height: "auto" }}
                      exit={{ opacity: 0, x: -30, height: 0 }}
                      transition={{ duration: 0.25, ease: "easeInOut" }}
                      className="hover:bg-surface-low/30 border-b border-surface-container overflow-hidden"
                    >
                      <td className="p-5 whitespace-nowrap">
                        <span className="block text-base font-bold text-primary">{item.nama}</span>
                        <span className="block text-xs font-mono text-outline mt-0.5 font-bold">ID: {item.id}</span>
                      </td>
                      <td className="p-5 whitespace-nowrap">
                        <span className="block text-base font-bold text-on-surface">{item.labor}</span>
                        <span className="block text-sm text-outline mt-0.5">Jurusan: {item.jurusan}</span>
                      </td>
                      <td className="p-5 text-center font-bold text-base tabular-nums whitespace-nowrap">{item.jumlah} Pcs</td>
                      <td className="p-5 text-base font-bold text-on-surface-variant whitespace-nowrap">{item.kategori}</td>
                      <td className="p-5 text-center whitespace-nowrap">
                        <span className={`inline-block px-3 py-1 rounded-md text-sm font-bold border ${
                          item.status === 'Baik' ? 'bg-green-100 text-green-900 border-green-300' :
                          item.status === 'Perbaikan' ? 'bg-amber-100 text-amber-900 border-amber-300' : 'bg-red-100 text-red-900 border-red-300'
                        }`}>
                          {item.status}
                        </span>
                      </td>
                      <td className="p-5 text-right space-x-2 whitespace-nowrap">
                        <button 
                          onClick={() => handleEditClick(item)}
                          className="p-2.5 border border-surface-container text-outline hover:text-primary hover:bg-secondary-container rounded-lg transition-all cursor-pointer inline-flex items-center"
                          title="Edit Barang"
                        >
                          <Edit2 className="w-4 h-4" />
                        </button>
                        <button 
                          onClick={() => handleDelete(item.id)}
                          className="p-2.5 border border-surface-container text-outline hover:text-error hover:bg-error-container rounded-lg transition-all cursor-pointer inline-flex items-center"
                          title="Hapus Barang"
                        >
                          <Trash2 className="w-4 h-4" />
                        </button>
                      </td>
                    </motion.tr>
                  ))}
                </AnimatePresence>
              </tbody>
            </table>
          </div>
        </div>

        {/* MODAL INPUT BARANG BARU */}
        <AnimatePresence>
          {isOpenAddModal && (
            <div className="fixed inset-0 bg-on-surface/40 backdrop-blur-sm z-50 flex items-center justify-center p-4">
              <motion.div 
                initial={{ opacity: 0, scale: 0.96 }}
                animate={{ opacity: 1, scale: 1 }}
                exit={{ opacity: 0, scale: 0.96 }}
                className="bg-white w-full max-w-xl border border-surface-container-high rounded-xl shadow-xl p-6 space-y-4 animate-in fade-in zoom-in-95 duration-150"
              >
                <div className="flex justify-between items-center border-b border-surface-container pb-2">
                  <h3 className="text-lg font-bold text-on-surface">Input Data Barang Labor</h3>
                  <button onClick={() => setIsOpenAddModal(false)} className="text-outline hover:text-on-surface p-1.5 rounded-lg border border-surface-container cursor-pointer"><X className="w-5 h-5" /></button>
                </div>
                <div className="space-y-4">
                  <div className="space-y-1"><label className="text-xs font-bold text-outline uppercase tracking-wider">Nama Barang</label><input type="text" placeholder="Contoh: Router Mikrotik RB951" className="w-full px-4 py-3 border border-surface-container-high rounded-lg text-base bg-white text-on-surface focus:outline-none focus:border-primary font-medium"/></div>
                  <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
                    <div className="space-y-1"><label className="text-xs font-bold text-outline uppercase tracking-wider">Jurusan</label><select className="w-full px-4 py-3 border border-surface-container-high rounded-lg text-base bg-white text-on-surface font-semibold"><option>PPLG</option><option>DKV</option><option>TKJ</option></select></div>
                    <div className="space-y-1"><label className="text-xs font-bold text-outline uppercase tracking-wider">Labor</label><select className="w-full px-4 py-3 border border-surface-container-high rounded-lg text-base bg-white text-on-surface font-semibold"><option>Labor Komputer 1</option><option>Labor Game Dev</option></select></div>
                  </div>
                  <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
                    <div className="space-y-1"><label className="text-xs font-bold text-outline uppercase tracking-wider">Jumlah Stok</label><input type="number" placeholder="0" className="w-full px-4 py-3 border border-surface-container-high rounded-lg text-base bg-white text-on-surface focus:outline-none focus:border-primary font-semibold"/></div>
                    <div className="space-y-1"><label className="text-xs font-bold text-outline uppercase tracking-wider">Kategori</label><select className="w-full px-4 py-3 border border-surface-container-high rounded-lg text-base bg-white text-on-surface font-semibold"><option>Perangkat Keras</option><option>Networking</option></select></div>
                  </div>
                </div>
                <div className="flex justify-end gap-3 pt-3 border-t border-surface-container">
                  <button onClick={() => setIsOpenAddModal(false)} className="px-5 py-2.5 text-sm font-bold text-secondary hover:bg-surface-low rounded-lg cursor-pointer">Batal</button>
                  <button onClick={() => setIsOpenAddModal(false)} className="px-6 py-2.5 text-sm font-bold text-white bg-primary hover:bg-primary-container rounded-lg cursor-pointer shadow-sm">Simpan Aset</button>
                </div>
              </motion.div>
            </div>
          )}
        </AnimatePresence>

        {/* MODAL EDIT BARANG PRO */}
        <AnimatePresence>
          {isOpenEditModal && selectedItem && (
            <div className="fixed inset-0 bg-on-surface/40 backdrop-blur-sm z-50 flex items-center justify-center p-4">
              <motion.div 
                initial={{ opacity: 0, scale: 0.96 }}
                animate={{ opacity: 1, scale: 1 }}
                exit={{ opacity: 0, scale: 0.96 }}
                className="bg-white w-full max-w-xl border border-surface-container-high rounded-xl shadow-xl p-6 space-y-4 animate-in fade-in zoom-in-95 duration-150"
              >
                <div className="flex justify-between items-center border-b border-surface-container pb-2">
                  <h3 className="text-lg font-bold text-on-surface">Modify Asset Parameters</h3>
                  <button onClick={() => setIsOpenEditModal(false)} className="text-outline hover:text-on-surface p-1.5 rounded-lg border border-surface-container cursor-pointer"><X className="w-5 h-5" /></button>
                </div>
                <div className="space-y-4">
                  <div className="space-y-1">
                    <label className="text-xs font-bold text-outline uppercase tracking-wider">Nama Barang</label>
                    <input type="text" defaultValue={selectedItem.nama} className="w-full px-4 py-3 border border-surface-container-high rounded-lg text-base bg-white text-on-surface focus:outline-none focus:border-primary font-medium"/>
                  </div>
                  <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
                    <div className="space-y-1">
                      <label className="text-xs font-bold text-outline uppercase tracking-wider">Jumlah Stok</label>
                      <input type="number" defaultValue={selectedItem.jumlah} className="w-full px-4 py-3 border border-surface-container-high rounded-lg text-base bg-white text-on-surface focus:outline-none focus:border-primary font-semibold"/>
                    </div>
                    <div className="space-y-1">
                      <label className="text-xs font-bold text-outline uppercase tracking-wider">Status Barang</label>
                      <select defaultValue={selectedItem.status} className="w-full px-4 py-3 border border-surface-container-high rounded-lg text-base bg-white text-on-surface focus:outline-none focus:border-primary font-semibold">
                        <option value="Baik">Baik</option>
                        <option value="Perbaikan">Perbaikan</option>
                        <option value="Rusak">Rusak</option>
                      </select>
                    </div>
                  </div>
                </div>
                <div className="flex justify-end gap-3 pt-3 border-t border-surface-container">
                  <button onClick={() => setIsOpenEditModal(false)} className="px-5 py-2.5 text-sm font-bold text-secondary hover:bg-surface-low rounded-lg cursor-pointer">Cancel</button>
                  <button onClick={() => setIsOpenEditModal(false)} className="px-6 py-2.5 text-sm font-bold text-white bg-primary hover:bg-primary-container rounded-lg cursor-pointer shadow-sm">Update Data</button>
                </div>
              </motion.div>
            </div>
          )}
        </AnimatePresence>
      </div>
    </PageAnimateWrapper>
  );
}
