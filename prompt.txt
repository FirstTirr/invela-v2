    setUsers(users.filter(u => u.id !== id));
  };

  return (
    <PageAnimateWrapper>
      <div className="space-y-6 relative">
        <div className="flex justify-between items-center border-b border-surface-container pb-4">
          <div>
            <h1 className="text-2xl font-bold tracking-tight text-on-surface">Memantau Akun Pengguna</h1>

          >
            <Plus className="w-4 h-4" /> Tambah Akun Baru
          </button>
        </div>

        {/* Table Monitoring */}
